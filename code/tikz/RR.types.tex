\newcommand{\teff}[1]{((\t>=#1)*exp(-(\t-#1)/6))}%
\newcommand{\ceff}[2]{((\t>=#1)*(\t-#1)-(\t>=#2)*(\t-#2))}%
\newcommand{\eplot}[4]{%
  \node[#2,lab,right] at (51,#1.5) {\strut#3 effect};
  \fill[ageplot,fill=#2,fade] (0,#1) -- plot (\t,{#1+#4}) -- (50,#1);
  \draw[ageplot,draw=#2]                plot (\t,{#1+#4}); }%
\newcommand{\class}[3]{%
  \def\y{#1}\def\ye{\y+.5}\def\ys{.67}
  \node[sec,anchor=west,rotate=90,minimum width=#2*4mm] at (-4,\y) {\strut#3}; }%
\begin{tikzpicture}[x=1.2mm,y=4mm]
  % fixed ----------------------------------------------------------------------
  \class{-2}{2}{fixed}
  \draw[red] node[lab,right] at (51,\ye) {exposure}
    (0,\ye) edge[-] (50,\ye);
  \eplot{\y+1}{blu}{lifetime}{\ys}
  % period ---------------------------------------------------------------------
  \class{-6.5}{4}{period}
  \draw[red] node[lab,right] at (51,\ye) {exposure}
    ( 7,\ye) edge[o-x] (15,\ye)
    (34,\ye) edge[o-]  (50,\ye);
  \eplot{\y+1}{grn}{transient} {\ys*or(and(\t>=7,\t<=15),and(\t>=34,\t<50))}
  \eplot{\y+2}{blu}{lifetime}  {\ys*(\t>=7)}
  \eplot{\y+3}{vio}{cumulative}{\ys/25*(\ceff{7}{15}+\ceff{34}{50})}
  % event ----------------------------------------------------------------------
